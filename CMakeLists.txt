cmake_minimum_required(VERSION 3.16)
project(warpsim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(warpsim_core src/sim.cpp)
target_include_directories(warpsim_core PUBLIC include)
add_executable(test_v0 tests/test_v0.cpp)
target_link_libraries(test_v0 PRIVATE warpsim_core)
enable_testing()
add_test(NAME v0 COMMAND test_v0)
